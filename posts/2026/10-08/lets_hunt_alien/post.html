Thank you EARL 2026

The talk is based on my blog back in Jun 2024, so most of the code will be there!

Functions To Help with Aggregation & Graphics 
agg.ufo_mart = function(df) {
    df |> 
    summarise(
        ufo_count = sum(ufo_count),
        ufo_multi_observe = sum(ufo_multi_observe),
        geometry = first(geometry),
        density = mean(as.numeric(DENSITY), na.rm=TRUE),
        population = mean(as.numeric(POP))
    ) |> 
    mutate(
        ufo_count_density = (ufo_count / density),
        ufo_multiserve_density = (ufo_multi_observe / density)
    )
}
collect_sf.ufo_mart =function(df) {
    x = df |> 
        collect() |> 
        mutate(
            geometry=sf::st_as_sfc(geometry, crs=4269)
        ) |> 
        sf::st_as_sf()
    class(x) <- c("ufo_mart", class(x))
    return(x)
}

as_ufo_mart=function(x) {
    if (inherits(x, "ufo_mart")) {
        return(x)
    } else {
        class(x) <- c("ufo_mart", class(c))
        return(x)
    }
}

vis_top_score.ufo_mart = function(x, 
    variable,top_n,desc=TRUE, 
    label_from = "full_name", 
    colorpalette = "brewer.bu_pu", 
    text_size=0.7,
    limits=c()
) {
    variable = enquo(variable)
    var_label = as_label(variable)
    if (length(limits) == 0) {
        limits = pull(x, {{variable}}) |> range()
    }
    x=x |> 
        arrange({{variable}} |> desc())
    if (desc) {
        x = x |> slice_head(n=top_n)
    } else {
        x = x |> slice_tail(n=top_n)
    }
    mid_point = pull(x, {{variable}}) |> median()
    label_glue = glue::glue("{{{label_from}}}")
    x |> 
        # transform
        mutate(
            "{var_label}" := round(!!variable,2),
            text_color=ifelse({{variable}} > mid_point, "white", "black"),
            label = glue::glue(label_glue)
        ) |> 
        sf::st_as_sf() |> 
        tmap::tm_shape() +
        tmap::tm_polygons(
            fill=var_label,
            fill.scale = tmap::tm_scale_continuous_log(values=colorpalette),
            fill.legend = tmap::tm_legend_hide(),
            fill_alpha=0.5,
            limits=limits
        ) +
        tmap::tm_text(
            text="label",
            col="text_color",
            size=text_size
        )
}
Function wraper for spdep local moran 
# Local Moran Statistics -----------------------
local_moran_c = function(variable, nb, get = c("Ii","E.Ii","Var.Ii","Pr(z > E(Ii))")) {
    lw = spdep::nb2listw(nb, style="W",zero.policy = TRUE)
    t = spdep::localmoran_perm( {{variable}}, lw, alternative="greater")
    t[,get]
}
local_moran_c_quadr= function(variable, nb, get = c("Ii","E.Ii","Var.Ii","Pr(z > E(Ii))")) {
    lw = spdep::nb2listw(nb, style="W",zero.policy = TRUE)
    t = spdep::localmoran_perm( {{variable}}, lw, alternative="greater")
    attributes(t)$quadr$mean
}
